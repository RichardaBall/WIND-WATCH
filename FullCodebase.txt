export class AudioManager {
    constructor() {
        this.startupAudio = new Audio('enginestartup.mp3');
        this.shutdownAudio = new Audio('engineshutdown.mp3');
        this.gearAudio = new Audio('gear.mp3');

        const AudioContext = window.AudioContext || window.webkitAudioContext;
        this.audioCtx = new AudioContext();
        this.rotorBuffer = null;
        this.rotorSource = null;

        // Auto-unlock audio context on user interaction
        const unlockAudio = () => {
            if (this.audioCtx && this.audioCtx.state === 'suspended') {
                this.audioCtx.resume();
            }
            if (this.audioCtx && this.audioCtx.state === 'running') {
                window.removeEventListener('pointerdown', unlockAudio);
                window.removeEventListener('keydown', unlockAudio);
            }
        };
        window.addEventListener('pointerdown', unlockAudio);
        window.addEventListener('keydown', unlockAudio);

        // Load continuous rotor sound buffer
        fetch('rotorsound.mp3')
            .then(response => response.arrayBuffer())
            .then(arrayBuffer => this.audioCtx.decodeAudioData(arrayBuffer))
            .then(buffer => {
                this.rotorBuffer = buffer;
            })
            .catch(e => console.warn("Error loading rotor sound buffer:", e));
    }

    playStartup(onEnded) {
        if (this.audioCtx && this.audioCtx.state === 'suspended') this.audioCtx.resume();
        this.startupAudio.onended = onEnded;
        this.startupAudio.currentTime = 0;
        this.startupAudio.play().catch(e => console.log("Startup audio play prevented:", e));
    }

    playShutdown(onEnded) {
        if (this.audioCtx && this.audioCtx.state === 'suspended') this.audioCtx.resume();
        this.shutdownAudio.onended = onEnded;
        this.shutdownAudio.currentTime = 0;
        this.shutdownAudio.play().catch(e => console.log("Shutdown audio play prevented:", e));
    }

    playGear() {
        if (this.audioCtx && this.audioCtx.state === 'suspended') this.audioCtx.resume();
        this.gearAudio.currentTime = 0;
        this.gearAudio.play().catch(e => console.log("Gear audio play prevented:", e));
    }

    playExplosion() {
        this.stopAllAudio();

        try {
            if (this.audioCtx) {
                if (this.audioCtx.state === 'suspended') {
                    this.audioCtx.resume();
                }

                const now = this.audioCtx.currentTime;
                const duration = 2.0;

                const bufferSize = this.audioCtx.sampleRate * duration;
                const buffer = this.audioCtx.createBuffer(1, bufferSize, this.audioCtx.sampleRate);
                const data = buffer.getChannelData(0);
                for (let i = 0; i < bufferSize; i++) {
                    data[i] = Math.random() * 2 - 1;
                }

                const noise = this.audioCtx.createBufferSource();
                noise.buffer = buffer;

                const filter = this.audioCtx.createBiquadFilter();
                filter.type = 'lowpass';
                filter.frequency.setValueAtTime(800, now);
                filter.frequency.exponentialRampToValueAtTime(40, now + duration);

                const gain = this.audioCtx.createGain();
                gain.gain.setValueAtTime(1.0, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + duration);

                noise.connect(filter);
                filter.connect(gain);
                gain.connect(this.audioCtx.destination);

                noise.start(now);
                noise.stop(now + duration);
            }
        } catch (e) {
            console.warn("Procedural explosion audio playback error:", e);
        }
    }

    playSplash() {
        this.stopAllAudio();

        try {
            if (this.audioCtx) {
                if (this.audioCtx.state === 'suspended') {
                    this.audioCtx.resume();
                }

                const now = this.audioCtx.currentTime;
                const duration = 1.8;

                const bufferSize = this.audioCtx.sampleRate * duration;
                const buffer = this.audioCtx.createBuffer(1, bufferSize, this.audioCtx.sampleRate);
                const data = buffer.getChannelData(0);
                for (let i = 0; i < bufferSize; i++) {
                    data[i] = Math.random() * 2 - 1;
                }

                const noise = this.audioCtx.createBufferSource();
                noise.buffer = buffer;

                const filter = this.audioCtx.createBiquadFilter();
                filter.type = 'bandpass';
                filter.frequency.setValueAtTime(500, now);
                filter.frequency.exponentialRampToValueAtTime(120, now + duration);

                const gain = this.audioCtx.createGain();
                gain.gain.setValueAtTime(0.8, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + duration);

                noise.connect(filter);
                filter.connect(gain);
                gain.connect(this.audioCtx.destination);

                noise.start(now);
                noise.stop(now + duration);
            }
        } catch (e) {
            console.warn("Procedural splash audio playback error:", e);
        }
    }

    stopAllAudio() {
        this.stopRotor();
        this.startupAudio.pause();
        this.startupAudio.currentTime = 0;
        this.shutdownAudio.pause();
        this.shutdownAudio.currentTime = 0;
        this.gearAudio.pause();
        this.gearAudio.currentTime = 0;
    }

    startRotor() {
        if (this.audioCtx && this.rotorBuffer && !this.rotorSource) {
            if (this.audioCtx.state === 'suspended') {
                this.audioCtx.resume();
            }
            this.rotorSource = this.audioCtx.createBufferSource();
            this.rotorSource.buffer = this.rotorBuffer;
            this.rotorSource.loop = true;
            this.rotorSource.connect(this.audioCtx.destination);
            this.rotorSource.start(0);
        }
    }

    stopRotor() {
        if (this.rotorSource) {
            try {
                this.rotorSource.stop();
            } catch (e) {}
            this.rotorSource.disconnect();
            this.rotorSource = null;
        }
    }

    getStartupDuration() {
        return this.startupAudio.duration;
    }

    getStartupCurrentTime() {
        return this.startupAudio.currentTime;
    }

    getShutdownDuration() {
        return this.shutdownAudio.duration;
    }

    getShutdownCurrentTime() {
        return this.shutdownAudio.currentTime;
    }
}

export class DayNightCycle {
    constructor(scene, sunLight, ambientLight, water) {
        this.scene = scene;
        this.sunLight = sunLight;
        this.ambientLight = ambientLight;
        this.water = water;

        this.timeOfDay = 0.25; 
        this.timeSpeed = 0.02; 
        this.paused = false;

        this.daySky = new THREE.Color(0x020d1a);
        this.nightSky = new THREE.Color(0x010408);
        
        this.dayWater = new THREE.Color(0x001e36);
        this.nightWater = new THREE.Color(0x00060d);
    }

    setTimeOfDay(val) {
        this.timeOfDay = val;
        this.paused = true;
    }

    update(delta) {
        if (!this.paused) {
            this.timeOfDay = (this.timeOfDay + delta * this.timeSpeed * 0.1) % 1.0;
        }

        const angle = this.timeOfDay * Math.PI * 2;
        const radius = 300;
        const sunX = Math.cos(angle) * radius;
        const sunY = Math.sin(angle) * radius;
        const sunZ = Math.sin(angle * 0.5) * 50;

        this.sunLight.position.set(sunX, sunY, sunZ);

        const sunHeightFactor = Math.sin(angle); 
        const nightFactor = THREE.MathUtils.clamp(-sunHeightFactor, 0.0, 1.0);

        this.sunLight.intensity = Math.max(0.0, sunHeightFactor) * 1.5;
        this.ambientLight.intensity = THREE.MathUtils.lerp(0.8, 0.15, nightFactor);

        this.scene.background.lerpColors(this.daySky, this.nightSky, nightFactor);
        if (this.scene.fog) {
            this.scene.fog.color.copy(this.scene.background);
        }

        if (this.water && this.water.material.uniforms['waterColor']) {
            const currentWaterColor = new THREE.Color().lerpColors(this.dayWater, this.nightWater, nightFactor);
            this.water.material.uniforms['waterColor'].value.copy(currentWaterColor);
        }

        return nightFactor;
    }
}

const clock = new THREE.Clock();

function animate() {
    requestAnimationFrame(animate);
    const delta = clock.getDelta();

    const nightFactor = dayNightCycle.update(delta);
    baseMaterial.color.setRGB(1 - nightFactor * 0.4, 1 - nightFactor * 0.4, 1 - nightFactor * 0.3);

    if (water.material.uniforms['time']) {
        water.material.uniforms['time'].value += delta * 0.3;
    }

    player.update(delta);

    const playerPos = player.getPosition();
    water.position.x = playerPos.x;
    water.position.z = playerPos.z;

    camera.position.x = playerPos.x + 50;
    camera.position.z = playerPos.z + 50;
    camera.lookAt(playerPos.x, 0, playerPos.z);

    baseMesh.quaternion.copy(camera.quaternion);
    player.mesh.quaternion.copy(camera.quaternion);

    renderer.render(scene, camera);
}

import * as THREE from 'three';

export class HelicopterPlayer {
    constructor(scene) {
        this.scene = scene;
        this.group = new THREE.Group();
        scene.add(this.group);

        const textureLoader = new THREE.TextureLoader();
        this.totalFrames = 8;

        const geometry = new THREE.PlaneGeometry(24, 24);
        
        this.material = new THREE.MeshStandardMaterial({
            color: 0xffffff,
            transparent: true,
            alphaTest: 0.5,
            roughness: 0.4,
            side: THREE.DoubleSide
        });

        this.mesh = new THREE.Mesh(geometry, this.material);
        this.group.add(this.mesh);

        // --- Rotor Group Permanent Settings ---
        this.rotorGroup = new THREE.Group();
        this.rotorGroup.position.set(0, 2.0, 0); 
        this.group.add(this.rotorGroup);

        const rotorGeo = new THREE.PlaneGeometry(8, 0.6);
        const rotorMat = new THREE.MeshStandardMaterial({
            color: 0x111111,
            roughness: 0.5,
            transparent: true,
            opacity: 0.85,
            side: THREE.DoubleSide
        });

        this.rotor1 = new THREE.Mesh(rotorGeo, rotorMat);
        this.rotor1.rotation.x = -Math.PI / 2;
        this.rotor1.scale.set(1.0, 0.6, 1.0); 
        this.rotorGroup.add(this.rotor1);

        this.rotor2 = new THREE.Mesh(rotorGeo, rotorMat);
        this.rotor2.rotation.x = -Math.PI / 2;
        this.rotor2.rotation.z = Math.PI / 2;
        this.rotor2.scale.set(1.0, 0.6, 1.0);
        this.rotorGroup.add(this.rotor2);

        this.rotorBlades = [this.rotor1, this.rotor2];
        this.rotorSpeed = 0;
        this.maxRotorSpeed = 90.0;

        this.gearDown = true;

        textureLoader.load('helicopter_gear.png', (tex) => {
            tex.colorSpace = THREE.SRGBColorSpace;
            tex.minFilter = THREE.LinearFilter;
            tex.magFilter = THREE.LinearFilter;
            tex.wrapS = THREE.RepeatWrapping;
            tex.wrapT = THREE.RepeatWrapping;
            tex.repeat.set(1 / this.totalFrames, 1);
            tex.offset.x = 0;

            this.gearDownTex = tex; 
            if (this.gearDown) {
                this.material.map = tex;
                this.material.needsUpdate = true;
            }
        }, undefined, (err) => {
            console.error('Error loading helicopter_gear.png:', err);
        });

        textureLoader.load('helicopter.png', (tex) => {
            tex.colorSpace = THREE.SRGBColorSpace;
            tex.minFilter = THREE.LinearFilter;
            tex.magFilter = THREE.LinearFilter;
            tex.wrapS = THREE.RepeatWrapping;
            tex.wrapT = THREE.RepeatWrapping;
            tex.repeat.set(1 / this.totalFrames, 1);
            tex.offset.x = 0;

            this.gearUpTex = tex; 
        }, undefined, (err) => {
            console.error('Error loading helicopter.png:', err);
        });

        this.group.position.set(0, 6, 0);
        this.targetPosition = new THREE.Vector3(0, 6, 0);
        this.speed = 25.0;
        this.isMoving = false;
        this.currentFrame = 3;

        this.engineState = 'OFF'; 
        this.targetAltitude = 6;  

        this.audioManager = new AudioManager();
        
        // Initialize Crash Handler
        this.crashHandler = new crash(scene);
        this.crashHandler.hideRestart();
        this.crashHandler.showRestart(() => {
            this.resetFlight();
        });
        this.crashHandler.hideRestart();

        this.hasCrashed = false;
        this.flightStarted = false;
    }

    resetFlight() {
        this.hasCrashed = false;
        this.flightStarted = false;
        this.engineState = 'OFF';
        this.gearDown = true;
        this.material.map = this.gearDownTex;
        this.material.needsUpdate = true;
        this.group.position.set(0, 6, 0);
        this.targetPosition.set(0, 6, 0);
        this.targetAltitude = 6;
        this.isMoving = false;
        this.rotorSpeed = 0;
        this.audioManager.stopAllAudio();
        this.crashHandler.hideRestart();
    }

    triggerCrash(isOceanCrash = false) {
        if (this.hasCrashed) return;
        this.hasCrashed = true;
        this.engineState = 'CRASHED';
        this.isMoving = false;
        
        if (isOceanCrash) {
            this.audioManager.playSplash();
        } else {
            this.audioManager.playExplosion();
        }

        this.crashHandler.deploy(this.group.position);
    }

    handleEngineKey() {
        if (this.hasCrashed) return;

        const padCenter = new THREE.Vector2(0, 0);
        const pos2D = new THREE.Vector2(this.group.position.x, this.group.position.z);
        const isOverOcean = pos2D.distanceTo(padCenter) > 10.0;

        if (this.engineState === 'OFF') {
            this.engineState = 'STARTING';
            this.targetPosition.copy(this.group.position);
            this.isMoving = false;
            this.flightStarted = true;
            
            this.audioManager.playStartup(() => {
                if (this.engineState === 'STARTING') {
                    this.engineState = 'RUNNING';
                    this.audioManager.startRotor();
                    this.targetAltitude = 16; 
                }
            });
        } else if (this.engineState === 'RUNNING') {
            this.engineState = 'SHUTTING_DOWN';
            this.targetPosition.copy(this.group.position);
            this.isMoving = false;
            this.audioManager.stopRotor();

            this.targetAltitude = isOverOcean ? 0.0 : 6.0; 
            
            this.audioManager.playShutdown(() => {
                if (this.engineState === 'SHUTTING_DOWN') {
                    this.engineState = 'OFF';
                }
            });
        }
    }

    setTarget(newPos) {
        if (this.hasCrashed || this.engineState !== 'RUNNING' || Math.abs(this.group.position.y - 16) > 1.0) {
            return;
        }

        const padCenter = new THREE.Vector2(0, 0);
        const clickPos2D = new THREE.Vector2(newPos.x, newPos.z);
        
        const snapThreshold = 8.0;
        if (clickPos2D.distanceTo(padCenter) < snapThreshold) {
            this.targetPosition.set(0, 16, 0);
        } else {
            this.targetPosition.copy(newPos);
            this.targetPosition.y = 16; 
        }

        this.isMoving = true;
    }

    toggleGear() {
        if (this.hasCrashed) return;

        if (this.gearDown && this.group.position.y < 10.0) {
            return;
        }

        this.gearDown = !this.gearDown;
        const currentOffset = this.material.map ? this.material.map.offset.x : 0;

        if (this.gearDown && this.gearDownTex) {
            this.material.map = this.gearDownTex;
        } else if (!this.gearDown && this.gearUpTex) {
            this.material.map = this.gearUpTex;
        }
        if (this.material.map) {
            this.material.map.offset.x = currentOffset;
        }
        this.material.needsUpdate = true;

        this.audioManager.playGear();
    }

    update(delta) {
        if (this.hasCrashed) return;

        const padCenter = new THREE.Vector2(0, 0);
        const pos2D = new THREE.Vector2(this.group.position.x, this.group.position.z);
        const distanceFromBase = pos2D.distanceTo(padCenter);
        const isOverOcean = distanceFromBase > 10.0;

        if (this.flightStarted) {
            if (isOverOcean && this.group.position.y <= 1.0) {
                this.triggerCrash(true);
                return;
            }

            if (this.engineState === 'SHUTTING_DOWN' && Math.abs(this.group.position.y - 6.0) < 0.5) {
                if (distanceFromBase < 8.0 && !this.gearDown) {
                    this.triggerCrash(false);
                    return;
                }
            }
        }

        if (this.engineState === 'STARTING') {
            const duration = this.audioManager.getStartupDuration();
            const currentTime = this.audioManager.getStartupCurrentTime();
            if (duration && !isNaN(duration)) {
                const progress = Math.min(1.0, currentTime / duration);
                this.rotorSpeed = progress * this.maxRotorSpeed;
            } else {
                this.rotorSpeed = Math.min(this.maxRotorSpeed, this.rotorSpeed + delta * 30);
            }
        } else if (this.engineState === 'RUNNING') {
            this.rotorSpeed = this.maxRotorSpeed;
        } else if (this.engineState === 'SHUTTING_DOWN') {
            const duration = this.audioManager.getShutdownDuration();
            const currentTime = this.audioManager.getShutdownCurrentTime();
            if (duration && !isNaN(duration)) {
                const progress = Math.max(0.0, 1.0 - (currentTime / duration));
                this.rotorSpeed = progress * this.maxRotorSpeed;
            } else {
                this.rotorSpeed = Math.max(0.0, this.rotorSpeed - delta * 30);
            }
        } else {
            this.rotorSpeed = 0;
        }

        for (const blade of this.rotorBlades) {
            blade.rotation.z += this.rotorSpeed * delta;
        }

        const currentPos = this.group.position;
        currentPos.y = THREE.MathUtils.lerp(currentPos.y, this.targetAltitude, delta * 3.0);

        if (this.engineState !== 'RUNNING' || Math.abs(currentPos.y - 16) > 1.0) {
            this.targetPosition.x = currentPos.x;
            this.targetPosition.z = currentPos.z;
            this.isMoving = false;
        }

        if (this.isMoving && this.engineState === 'RUNNING') {
            const distance = new THREE.Vector2(currentPos.x, currentPos.z).distanceTo(new THREE.Vector2(this.targetPosition.x, this.targetPosition.z));

            if (distance > 0.5) {
                const dir = new THREE.Vector3().subVectors(this.targetPosition, currentPos);
                dir.y = 0;
                dir.normalize();
                
                currentPos.x += dir.x * this.speed * delta;
                currentPos.z += dir.z * this.speed * delta;

                const angle = Math.atan2(dir.z, dir.x) + Math.PI;
                let normalized = (angle + Math.PI * 2) % (Math.PI * 2);
                
                const frameIndex = Math.floor((normalized / (Math.PI * 2)) * this.totalFrames + 0.5) % this.totalFrames;
                this.currentFrame = frameIndex;

                if (this.material.map) {
                    this.material.map.offset.x = this.currentFrame * (1 / this.totalFrames);
                }
            } else {
                this.isMoving = false;
            }
        }
    }

    getPosition() {
        return this.group.position;
    }
}
import * as THREE from 'three';
import { Water } from 'three/addons/objects/Water.js';

export function setupScene() {
    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0x020d1a);
    scene.fog = new THREE.FogExp2(0x020d1a, 0.001);

    const aspect = window.innerWidth / window.innerHeight;
    const frustumSize = 80;
    
    const camera = new THREE.OrthographicCamera(
        (frustumSize * aspect) / -2, 
        (frustumSize * aspect) / 2, 
        frustumSize / 2, 
        frustumSize / -2, 
        0.1, 
        2000
    );

    camera.position.set(50, 60, 50);
    camera.lookAt(0, 0, 0);

    const renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance' });
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.0;
    document.body.appendChild(renderer.domElement);

    const ambientLight = new THREE.AmbientLight(0x6080a0, 0.8);
    scene.add(ambientLight);

    const sunLight = new THREE.DirectionalLight(0xffffff, 1.5);
    sunLight.position.set(100, 200, 100);
    scene.add(sunLight);

    const waterGeometry = new THREE.PlaneGeometry(5000, 5000, 64, 64);
    const waterNormals = new THREE.TextureLoader().load(
        'https://raw.githubusercontent.com/mrdoob/three.js/master/examples/textures/waternormals.jpg',
        (texture) => {
            texture.wrapS = texture.wrapT = THREE.RepeatWrapping;
            texture.repeat.set(30, 30);
        }
    );

    const water = new Water(waterGeometry, {
        textureWidth: 512,
        textureHeight: 512,
        waterNormals: waterNormals,
        sunDirection: sunLight.position.clone().normalize(),
        sunColor: 0x336699,
        waterColor: 0x001e36,
        distortionScale: 3.7,
        fog: scene.fog !== undefined
    });
    water.rotation.x = -Math.PI / 2;
    water.position.y = -2.0;
    scene.add(water);

    window.addEventListener('resize', () => {
        const newAspect = window.innerWidth / window.innerHeight;
        camera.left = (frustumSize * newAspect) / -2;
        camera.right = (frustumSize * newAspect) / 2;
        camera.top = frustumSize / 2;
        camera.bottom = frustumSize / -2;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
    });

    return { scene, camera, renderer, water, sunLight, ambientLight };
}
export function setupSettings(dayNightCycle) {
    const container = document.createElement('div');
    container.style.position = 'absolute';
    container.style.top = '20px';
    container.style.right = '20px';
    container.style.background = 'rgba(0, 0, 0, 0.85)';
    container.style.padding = '12px 15px';
    container.style.borderRadius = '6px';
    container.style.border = '1px solid rgba(255, 255, 255, 0.2)';
    container.style.color = '#ffffff';
    container.style.fontFamily = 'monospace';
    container.style.fontSize = '12px';
    container.style.zIndex = '100';

    const title = document.createElement('div');
    title.innerHTML = '<strong>Dev Settings</strong> (Press <strong>T</strong> to toggle)';
    title.style.marginBottom = '10px';
    container.appendChild(title);

    const toggleBtn = document.createElement('button');
    toggleBtn.innerText = 'Toggle Day / Night';
    toggleBtn.style.padding = '6px 12px';
    toggleBtn.style.background = '#1a365d';
    toggleBtn.style.color = '#ffffff';
    toggleBtn.style.border = '1px solid #4a90e2';
    toggleBtn.style.borderRadius = '4px';
    toggleBtn.style.cursor = 'pointer';
    container.appendChild(toggleBtn);

    let isNight = false;
    toggleBtn.onclick = () => {
        isNight = !isNight;
        dayNightCycle.setTimeOfDay(isNight ? 0.0 : 0.25);
    };

    document.body.appendChild(container);

    let isVisible = true;
    window.addEventListener('keydown', (e) => {
        if (e.code === 'KeyT') {
            isVisible = !isVisible;
            container.style.display = isVisible ? 'block' : 'none';
        }
    });
}
