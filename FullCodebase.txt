export class AudioManager {
    constructor() {
        this.startupAudio = new Audio('enginestartup.mp3');
        this.shutdownAudio = new Audio('engineshutdown.mp3');
        this.gearAudio = new Audio('gear.mp3');

        const AudioContext = window.AudioContext || window.webkitAudioContext;
        this.audioCtx = new AudioContext();
        this.rotorBuffer = null;
        this.rotorSource = null;

        // Auto-unlock audio context on user interaction
        const unlockAudio = () => {
            if (this.audioCtx && this.audioCtx.state === 'suspended') {
                this.audioCtx.resume();
            }
            if (this.audioCtx && this.audioCtx.state === 'running') {
                window.removeEventListener('pointerdown', unlockAudio);
                window.removeEventListener('keydown', unlockAudio);
            }
        };
        window.addEventListener('pointerdown', unlockAudio);
        window.addEventListener('keydown', unlockAudio);

        // Load continuous rotor sound buffer
        fetch('rotorsound.mp3')
            .then(response => response.arrayBuffer())
            .then(arrayBuffer => this.audioCtx.decodeAudioData(arrayBuffer))
            .then(buffer => {
                this.rotorBuffer = buffer;
            })
            .catch(e => console.warn("Error loading rotor sound buffer:", e));
    }

    playStartup(onEnded) {
        if (this.audioCtx && this.audioCtx.state === 'suspended') this.audioCtx.resume();
        this.startupAudio.onended = onEnded;
        this.startupAudio.currentTime = 0;
        this.startupAudio.play().catch(e => console.log("Startup audio play prevented:", e));
    }

    playShutdown(onEnded) {
        if (this.audioCtx && this.audioCtx.state === 'suspended') this.audioCtx.resume();
        this.shutdownAudio.onended = onEnded;
        this.shutdownAudio.currentTime = 0;
        this.shutdownAudio.play().catch(e => console.log("Shutdown audio play prevented:", e));
    }

    playGear() {
        if (this.audioCtx && this.audioCtx.state === 'suspended') this.audioCtx.resume();
        this.gearAudio.currentTime = 0;
        this.gearAudio.play().catch(e => console.log("Gear audio play prevented:", e));
    }

    playExplosion() {
        this.stopAllAudio();

        try {
            if (this.audioCtx) {
                if (this.audioCtx.state === 'suspended') {
                    this.audioCtx.resume();
                }

                const now = this.audioCtx.currentTime;
                const duration = 2.0;

                const bufferSize = this.audioCtx.sampleRate * duration;
                const buffer = this.audioCtx.createBuffer(1, bufferSize, this.audioCtx.sampleRate);
                const data = buffer.getChannelData(0);
                for (let i = 0; i < bufferSize; i++) {
                    data[i] = Math.random() * 2 - 1;
                }

                const noise = this.audioCtx.createBufferSource();
                noise.buffer = buffer;

                const filter = this.audioCtx.createBiquadFilter();
                filter.type = 'lowpass';
                filter.frequency.setValueAtTime(800, now);
                filter.frequency.exponentialRampToValueAtTime(40, now + duration);

                const gain = this.audioCtx.createGain();
                gain.gain.setValueAtTime(1.0, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + duration);

                noise.connect(filter);
                filter.connect(gain);
                gain.connect(this.audioCtx.destination);

                noise.start(now);
                noise.stop(now + duration);
            }
        } catch (e) {
            console.warn("Procedural explosion audio playback error:", e);
        }
    }

    playSplash() {
        this.stopAllAudio();

        try {
            if (this.audioCtx) {
                if (this.audioCtx.state === 'suspended') {
                    this.audioCtx.resume();
                }

                const now = this.audioCtx.currentTime;
                const duration = 1.8;

                const bufferSize = this.audioCtx.sampleRate * duration;
                const buffer = this.audioCtx.createBuffer(1, bufferSize, this.audioCtx.sampleRate);
                const data = buffer.getChannelData(0);
                for (let i = 0; i < bufferSize; i++) {
                    data[i] = Math.random() * 2 - 1;
                }

                const noise = this.audioCtx.createBufferSource();
                noise.buffer = buffer;

                const filter = this.audioCtx.createBiquadFilter();
                filter.type = 'bandpass';
                filter.frequency.setValueAtTime(500, now);
                filter.frequency.exponentialRampToValueAtTime(120, now + duration);

                const gain = this.audioCtx.createGain();
                gain.gain.setValueAtTime(0.8, now);
                gain.gain.exponentialRampToValueAtTime(0.001, now + duration);

                noise.connect(filter);
                filter.connect(gain);
                gain.connect(this.audioCtx.destination);

                noise.start(now);
                noise.stop(now + duration);
            }
        } catch (e) {
            console.warn("Procedural splash audio playback error:", e);
        }
    }

    stopAllAudio() {
        this.stopRotor();
        this.startupAudio.pause();
        this.startupAudio.currentTime = 0;
        this.shutdownAudio.pause();
        this.shutdownAudio.currentTime = 0;
        this.gearAudio.pause();
        this.gearAudio.currentTime = 0;
    }

    startRotor() {
        if (this.audioCtx && this.rotorBuffer && !this.rotorSource) {
            if (this.audioCtx.state === 'suspended') {
                this.audioCtx.resume();
            }
            this.rotorSource = this.audioCtx.createBufferSource();
            this.rotorSource.buffer = this.rotorBuffer;
            this.rotorSource.loop = true;
            this.rotorSource.connect(this.audioCtx.destination);
            this.rotorSource.start(0);
        }
    }

    stopRotor() {
        if (this.rotorSource) {
            try {
                this.rotorSource.stop();
            } catch (e) {}
            this.rotorSource.disconnect();
            this.rotorSource = null;
        }
    }

    getStartupDuration() {
        return this.startupAudio.duration;
    }

    getStartupCurrentTime() {
        return this.startupAudio.currentTime;
    }

    getShutdownDuration() {
        return this.shutdownAudio.duration;
    }

    getShutdownCurrentTime() {
        return this.shutdownAudio.currentTime;
    }
}

export class DayNightCycle {
    constructor(scene, sunLight, ambientLight, ocean) {
        this.scene = scene;
        this.sunLight = sunLight;
        this.ambientLight = ambientLight;
        this.ocean = ocean;

        this.timeOfDay = 0.25; // 0.25 = Noon, 0.75 = Midnight
        this.timeScale = 1.0; 
        this.paused = false;

        // Full cycle = 10 minutes (600 seconds) -> 5 mins day, 5 mins night
        this.baseSpeed = 1.0 / 600.0;

        this.daySky = new THREE.Color(0x020d1a);
        this.nightSky = new THREE.Color(0x010408);

        // Create visual Sun mesh
        const sunGeo = new THREE.SphereGeometry(12, 16, 16);
        const sunMat = new THREE.MeshBasicMaterial({ color: 0xffffee });
        this.sunMesh = new THREE.Mesh(sunGeo, sunMat);
        this.scene.add(this.sunMesh);

        // Create visual Moon mesh
        const moonGeo = new THREE.SphereGeometry(10, 16, 16);
        const moonMat = new THREE.MeshBasicMaterial({ color: 0x99ccff });
        this.moonMesh = new THREE.Mesh(moonGeo, moonMat);
        this.scene.add(this.moonMesh);
    }

    setTimeOfDay(val) {
        this.timeOfDay = val;
        this.paused = true;
    }

    toggleTimelapse() {
        this.timeScale = this.timeScale === 1.0 ? 100.0 : 1.0;
        this.paused = false; // Resume if it was paused by manual setting overrides
        return this.timeScale === 100.0;
    }

    update(delta) {
        if (!this.paused) {
            this.timeOfDay = (this.timeOfDay + delta * this.baseSpeed * this.timeScale) % 1.0;
        }

        const angle = this.timeOfDay * Math.PI * 2;
        const radius = 400;
        
        // Sun orbital position
        const sunX = Math.cos(angle) * radius;
        const sunY = Math.sin(angle) * radius;
        const sunZ = 50;
        this.sunLight.position.set(sunX, sunY, sunZ);
        this.sunMesh.position.set(sunX, sunY, sunZ);

        // Moon orbital position (opposite to sun)
        const moonX = Math.cos(angle + Math.PI) * radius;
        const moonY = Math.sin(angle + Math.PI) * radius;
        const moonZ = 50;
        this.moonMesh.position.set(moonX, moonY, moonZ);

        const sunHeightFactor = Math.sin(angle); 
        const nightFactor = THREE.MathUtils.clamp(-sunHeightFactor, 0.0, 1.0);

        this.sunLight.intensity = Math.max(0.0, sunHeightFactor) * 1.5;
        this.ambientLight.intensity = THREE.MathUtils.lerp(0.8, 0.15, nightFactor);

        this.scene.background.lerpColors(this.daySky, this.nightSky, nightFactor);
        if (this.scene.fog) {
            this.scene.fog.color.copy(this.scene.background);
        }

        // Update ocean night overlay
        if (this.ocean) {
            this.ocean.updateLighting(nightFactor);
        }

        // Hide celestial bodies when below horizon
        this.sunMesh.visible = sunY > -30;
        this.moonMesh.visible = moonY > -30;

        return nightFactor;
    }
}

export class HelicopterSystems {
    constructor(scene, helicopterGroup, playerRef) {
        this.scene = scene;
        this.helicopterGroup = helicopterGroup;
        this.playerRef = playerRef;

        this.tools = {
            WATER: new WaterSystem(scene, helicopterGroup, playerRef),
            WINCH: new WinchSystem(scene, helicopterGroup)
        };

        this.activeToolKey = 'WATER';
    }

    setTool(toolKey) {
        if (this.tools[toolKey]) {
            this.activeToolKey = toolKey;
            console.log(`Switched to tool: ${toolKey}`);
        }
    }

    startActiveTool() {
        const activeTool = this.tools[this.activeToolKey];
        if (activeTool && typeof activeTool.start === 'function') {
            activeTool.start();
        } else if (activeTool && typeof activeTool.toggle === 'function') {
            activeTool.toggle();
        }
    }

    stopActiveTool() {
        const activeTool = this.tools[this.activeToolKey];
        if (activeTool && typeof activeTool.stop === 'function') {
            activeTool.stop();
        }
    }

    toggleActiveTool() {
        const activeTool = this.tools[this.activeToolKey];
        if (activeTool && typeof activeTool.toggle === 'function') {
            activeTool.toggle();
        }
    }

    update(delta) {
        for (const key in this.tools) {
            if (this.tools[key].update) {
                this.tools[key].update(delta);
            }
        }
    }
}

const raycaster = new THREE.Raycaster();
const mouse = new THREE.Vector2();

// Raycast intersection plane at ground level (y = 0)
const planeIntersectTarget = new THREE.Plane(new THREE.Vector3(0, 1, 0), 0);

window.addEventListener('click', (e) => {
    mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
    mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;

    raycaster.setFromCamera(mouse, camera);
    const targetPoint = new THREE.Vector3();
    raycaster.ray.intersectPlane(planeIntersectTarget, targetPoint);

    if (targetPoint) {
        // Offset the target position upward by the flight altitude (32) 
        // so the shadow lands precisely on the clicked ground spot!
        targetPoint.y = 32;
        player.setTarget(targetPoint);
    }
});

function animate() {
    requestAnimationFrame(animate);
    const delta = clock.getDelta();

    const nightFactor = dayNightCycle.update(delta);

    player.update(delta);
    helicopterSystems.update(delta);

    const playerPos = player.getPosition();
    ocean.update(delta, playerPos);

    camera.position.x = playerPos.x + 50;
    camera.position.z = playerPos.z + 50;
    camera.position.y = playerPos.y + 50;
    camera.lookAt(playerPos.x, playerPos.y, playerPos.z);

    mainBase.update(nightFactor, camera.quaternion);
    windFarm.update(delta, camera.quaternion);
    
    player.mesh.quaternion.copy(camera.quaternion);

    renderer.render(scene, camera);
}

import * as THREE from 'three';

export class MainBase {
    constructor(scene, camera, renderer) {
        this.scene = scene;
        this.camera = camera;
        this.renderer = renderer;

        this.group = new THREE.Group();
        this.group.position.set(0, 5, 0);
        scene.add(this.group);

        const textureLoader = new THREE.TextureLoader();
        const baseTexture = textureLoader.load('mainbase.png', (tex) => {
            tex.colorSpace = THREE.SRGBColorSpace;
            tex.minFilter = THREE.LinearFilter;
        });

        this.material = new THREE.MeshStandardMaterial({
            map: baseTexture,
            transparent: true,
            alphaTest: 0.5,
            roughness: 0.4
        });

        const baseGeometry = new THREE.PlaneGeometry(24, 24);
        this.mesh = new THREE.Mesh(baseGeometry, this.material);
        this.group.add(this.mesh);

        // --- Exact Configured Foam Settings (Flat on XZ plane with correct rotation, offsets, and height) ---
        this.foamParams = {
            radius: 7.5,
            offsetX: -0.6,
            offsetZ: -1.0,
            posY: -2.9,
            rotX: -45.0,
            rotY: 0.0,
            rotZ: 0.0,
            speed: 0.2
        };

        this.particlesCount = 2400;
        this.foamGeometry = new THREE.BufferGeometry();
        this.foamPositions = new Float32Array(this.particlesCount * 3);
        this.foamAngles = new Float32Array(this.particlesCount);
        this.foamOffsets = new Float32Array(this.particlesCount);

        for (let i = 0; i < this.particlesCount; i++) {
            this.foamPositions[i * 3] = 0;
            this.foamPositions[i * 3 + 1] = 0;
            this.foamPositions[i * 3 + 2] = 0;
            this.foamAngles[i] = (i / this.particlesCount) * Math.PI * 2;
            this.foamOffsets[i] = Math.random() * Math.PI * 2;
        }

        this.foamGeometry.setAttribute('position', new THREE.BufferAttribute(this.foamPositions, 3));

        const canvas = document.createElement('canvas');
        canvas.width = 64;
        canvas.height = 64;
        const ctx = canvas.getContext('2d');
        const gradient = ctx.createRadialGradient(32, 32, 0, 32, 32, 32);
        gradient.addColorStop(0, 'rgba(255, 255, 255, 0.9)');
        gradient.addColorStop(0.4, 'rgba(240, 248, 255, 0.4)');
        gradient.addColorStop(1, 'rgba(255, 255, 255, 0)');
        ctx.fillStyle = gradient;
        ctx.fillRect(0, 0, 64, 64);

        const foamTexture = new THREE.CanvasTexture(canvas);

        this.foamMaterial = new THREE.PointsMaterial({
            color: 0xffffff,
            size: 10.0,
            map: foamTexture,
            transparent: true,
            blending: THREE.NormalBlending,
            depthWrite: false,
            opacity: 0.7
        });

        this.foamParticleSystem = new THREE.Points(this.foamGeometry, this.foamMaterial);
        this.foamParticleSystem.frustumCulled = false;
        this.foamParticleSystem.position.set(this.foamParams.offsetX, this.foamParams.posY, this.foamParams.offsetZ);
        this.foamParticleSystem.rotation.x = THREE.MathUtils.degToRad(this.foamParams.rotX);
        this.foamParticleSystem.rotation.y = THREE.MathUtils.degToRad(this.foamParams.rotY);
        this.foamParticleSystem.rotation.z = THREE.MathUtils.degToRad(this.foamParams.rotZ);
        this.group.add(this.foamParticleSystem);

        // --- Final Saved Light Layout Configurations ---
        this.lightConfigs = {
            green: [
                { x: -2.1, y: -1.4, scale: 0.05 },
                { x: 2, y: -1.5, scale: 0.05 },
                { x: -1.7, y: 1.2, scale: 0.05 },
                { x: 2, y: 1.1, scale: 0.05 }
            ],
            red: [
                { x: 0, y: -5, scale: 0.05 },
                { x: 6, y: 0, scale: 0.05 },
                { x: -6, y: 0, scale: 0.05 },
                { x: 0, y: 3.2, scale: 0.05 },
                { x: -10.1, y: 12, scale: 0.05 },
                { x: -10.7, y: 11.2, scale: 0.05 }
            ],
            white: [
                { x: -5, y: 2.5, scale: 0 },
                { x: 3, y: 3.5, scale: 0 }
            ]
        };

        this.allLights = [];
        this.buildLightGroup('green', 0x00ff66);
        this.buildLightGroup('red', 0xff0033);
        this.buildLightGroup('white', 0xffffff);
    }

    buildLightGroup(category, colorHex) {
        this.lightConfigs[category].forEach((pos, idx) => {
            const markerGeo = new THREE.SphereGeometry(1, 16, 16);
            const markerMat = new THREE.MeshBasicMaterial({ color: colorHex });
            const marker = new THREE.Mesh(markerGeo, markerMat);
            marker.position.set(pos.x, pos.y, 0.2);
            marker.scale.set(pos.scale, pos.scale, pos.scale);
            this.group.add(marker);

            const pointLight = new THREE.PointLight(colorHex, 0.0, 15);
            pointLight.position.set(pos.x, pos.y, 0.2);
            this.group.add(pointLight);

            this.allLights.push({ category, index: idx, marker, pointLight });
        });
    }

    update(nightFactor, cameraQuaternion) {
        this.material.color.setRGB(1 - nightFactor * 0.4, 1 - nightFactor * 0.4, 1 - nightFactor * 0.3);

        if (this.foamMaterial) {
            this.foamMaterial.opacity = nightFactor > 0.5 ? 0.25 : 0.7;
        }

        const elapsedTime = Date.now() * 0.001;

        for (let i = 0; i < this.particlesCount; i++) {
            const idx = i * 3;
            const angle = this.foamAngles[i];
            const phase = this.foamOffsets[i];

            const surge = Math.sin(elapsedTime * this.foamParams.speed + phase) * 0.8;
            const currentRadius = this.foamParams.radius + surge;

            const x = Math.cos(angle) * currentRadius;
            const y = Math.sin(angle) * currentRadius;
            const z = Math.sin(elapsedTime * 3.0 + i) * 0.08;

            this.foamPositions[idx] = x;
            this.foamPositions[idx + 1] = y;
            this.foamPositions[idx + 2] = z;
        }

        this.foamGeometry.attributes.position.needsUpdate = true;

        const intensityMultiplier = nightFactor * 3.0;
        this.allLights.forEach(item => {
            item.pointLight.intensity = intensityMultiplier;
        });

        if (cameraQuaternion) {
            this.group.quaternion.copy(cameraQuaternion);
        }
    }
}

export class OceanWater {
    constructor(scene, sunLight) {
        const waterGeometry = new THREE.PlaneGeometry(5000, 5000, 64, 64);
        const waterNormals = new THREE.TextureLoader().load(
            'https://raw.githubusercontent.com/mrdoob/three.js/master/examples/textures/waternormals.jpg',
            (texture) => {
                texture.wrapS = texture.wrapT = THREE.RepeatWrapping;
                texture.repeat.set(30, 30);
            }
        );

        // Pristine original day water configuration
        this.water = new Water(waterGeometry, {
            textureWidth: 512,
            textureHeight: 512,
            waterNormals: waterNormals,
            sunDirection: sunLight.position.clone().normalize(),
            sunColor: 0x336699,
            waterColor: 0x001e36,
            distortionScale: 3.7,
            fog: scene.fog !== undefined
        });
        
        this.water.rotation.x = -Math.PI / 2;
        this.water.position.y = -2.0;
        scene.add(this.water);

        // Subtle dark overlay plane just above the water for smooth night dimming
        const overlayGeometry = new THREE.PlaneGeometry(5000, 5000);
        this.nightOverlayMaterial = new THREE.MeshBasicMaterial({
            color: 0x000208,
            transparent: true,
            opacity: 0.0, // Fully transparent during day
            depthWrite: false
        });
        this.nightOverlay = new THREE.Mesh(overlayGeometry, this.nightOverlayMaterial);
        this.nightOverlay.rotation.x = -Math.PI / 2;
        this.nightOverlay.position.y = -1.9; // Just above water surface
        scene.add(this.nightOverlay);
    }

    updateLighting(nightFactor) {
        // Smoothly fade in the dark overlay at night without breaking the water shader
        if (this.nightOverlayMaterial) {
            this.nightOverlayMaterial.opacity = nightFactor * 0.75;
        }
    }

    update(delta, playerPosition) {
        if (this.water.material.uniforms['time']) {
            this.water.material.uniforms['time'].value += delta * 0.3;
        }
        this.water.position.x = playerPosition.x;
        this.water.position.z = playerPosition.z;

        this.nightOverlay.position.x = playerPosition.x;
        this.nightOverlay.position.z = playerPosition.z;
    }
}

export class HelicopterPlayer {

            tex.colorSpace = THREE.SRGBColorSpace;
            tex.minFilter = THREE.LinearFilter;
            tex.magFilter = THREE.LinearFilter;
            tex.wrapS = THREE.RepeatWrapping;
            tex.wrapT = THREE.RepeatWrapping;
            tex.repeat.set(1 / this.totalFrames, 1);
            tex.offset.x = this.currentFrame * (1 / this.totalFrames);

            this.emissiveTex = tex;
            this.material.emissiveMap = tex;
            this.material.needsUpdate = true;
        }, undefined, (err) => {
            console.warn('helicopter_emissive.png not found.');
        });

        this.group.position.set(0, 6, 0);
        this.targetPosition = new THREE.Vector3(0, 6, 0);
        this.speed = 25.0;
        this.isMoving = false;

        this.engineState = 'OFF'; 
        this.targetAltitude = 6;  

        this.audioManager = new AudioManager();
        
        this.crashHandler = new crash(scene);
        this.crashHandler.hideRestart();
        this.crashHandler.showRestart(() => {
            this.resetFlight();
        });
        this.crashHandler.hideRestart();

        this.hasCrashed = false;
        this.flightStarted = false;
    }

    setVisibility(visible) {
        if (this.group) {
            this.group.visible = visible;
            if (this.shadowMesh) this.shadowMesh.visible = visible;
            if (this.dropLine) this.dropLine.visible = visible;
        }
    }

    resetFlight() {
        this.hasCrashed = false;
        this.flightStarted = false;
        this.engineState = 'OFF';
        this.gearDown = true;
        this.currentFrame = 5;
        if (this.gearDownTex && this.material) {
            this.material.map = this.gearDownTex;
            this.material.map.offset.x = 5 * (1 / this.totalFrames);
            this.material.needsUpdate = true;
        }
        if (this.emissiveTex && this.material) {
            this.material.emissiveMap.offset.x = 5 * (1 / this.totalFrames);
            this.material.needsUpdate = true;
        }
        this.group.position.set(0, 6, 0);
        this.targetPosition.set(0, 6, 0);
        this.targetAltitude = 6;
        this.isMoving = false;
        this.rotorSpeed = 0;
        this.audioManager.stopAllAudio();
        this.crashHandler.hideRestart();
        if (this.shadowMesh) this.shadowMesh.visible = true;
        if (this.dropLine) this.dropLine.visible = true;
    }

    triggerCrash(isOceanCrash = false) {
        if (this.hasCrashed) return;
        this.hasCrashed = true;
        this.engineState = 'CRASHED';
        this.isMoving = false;
        if (this.shadowMesh) this.shadowMesh.visible = false;
        if (this.dropLine) this.dropLine.visible = false;
        
        if (isOceanCrash) {
            this.audioManager.playSplash();
        } else {
            this.audioManager.playExplosion();
        }

        this.crashHandler.deploy(this.group.position);
    }

    handleEngineKey() {
        if (this.hasCrashed) return;

        const padCenter = new THREE.Vector2(0, 0);
        const pos2D = new THREE.Vector2(this.group.position.x, this.group.position.z);
        const isOverOcean = pos2D.distanceTo(padCenter) > 10.0;

        if (this.engineState === 'OFF') {
            this.engineState = 'STARTING';
            this.targetPosition.copy(this.group.position);
            this.isMoving = false;
            this.flightStarted = true;
            
            this.audioManager.playStartup(() => {
                if (this.engineState === 'STARTING') {
                    this.engineState = 'RUNNING';
                    this.audioManager.startRotor();
                    this.targetAltitude = 32; 
                }
            });
        } else if (this.engineState === 'RUNNING') {
            this.engineState = 'SHUTTING_DOWN';
            this.targetPosition.copy(this.group.position);
            this.isMoving = false;
            this.audioManager.stopRotor();

            this.targetAltitude = isOverOcean ? 0.0 : 6.0; 
            
            this.audioManager.playShutdown(() => {
                if (this.engineState === 'SHUTTING_DOWN') {
                    this.engineState = 'OFF';
                }
            });
        }
    }

    setTarget(newPos) {
        if (this.hasCrashed || this.engineState !== 'RUNNING' || Math.abs(this.group.position.y - 32) > 1.0) {
            return;
        }

        const padCenter = new THREE.Vector2(0, 0);
        const clickPos2D = new THREE.Vector2(newPos.x, newPos.z);
        
        const snapThreshold = 8.0;
        if (clickPos2D.distanceTo(padCenter) < snapThreshold) {
            this.targetPosition.set(0, 32, 0);
        } else {
            this.targetPosition.copy(newPos);
            this.targetPosition.y = 32; 
        }

        this.isMoving = true;
    }

    toggleGear() {
        if (this.hasCrashed) return;

        if (this.gearDown && this.group.position.y < 10.0) {
            return;
        }

        this.gearDown = !this.gearDown;
        const currentOffset = this.material.map ? this.material.map.offset.x : 0;

        if (this.gearDown && this.gearDownTex) {
            this.material.map = this.gearDownTex;
        } else if (!this.gearDown && this.gearUpTex) {
            this.material.map = this.gearUpTex;
        }
        if (this.material.map) {
            this.material.map.offset.x = currentOffset;
        }
        this.material.needsUpdate = true;

        this.audioManager.playGear();
    }

    update(delta) {
        if (this.hasCrashed) return;

        const padCenter = new THREE.Vector2(0, 0);
        const pos2D = new THREE.Vector2(this.group.position.x, this.group.position.z);
        const distanceFromBase = pos2D.distanceTo(padCenter);
        const isOverOcean = distanceFromBase > 10.0;

        if (this.flightStarted) {
            if (isOverOcean && this.group.position.y <= 1.0) {
                this.triggerCrash(true);
                return;
            }

            if (this.engineState === 'SHUTTING_DOWN' && Math.abs(this.group.position.y - 6.0) < 0.5) {
                if (distanceFromBase < 8.0 && !this.gearDown) {
                    this.triggerCrash(false);
                    return;
                }
            }
        }

        if (this.engineState === 'STARTING') {
            const duration = this.audioManager.getStartupDuration();
            const currentTime = this.audioManager.getStartupCurrentTime();
            if (duration && !isNaN(duration)) {
                const progress = Math.min(1.0, currentTime / duration);
                this.rotorSpeed = progress * this.maxRotorSpeed;
            } else {
                this.rotorSpeed = Math.min(this.maxRotorSpeed, this.rotorSpeed + delta * 30);
            }
        } else if (this.engineState === 'RUNNING') {
            this.rotorSpeed = this.maxRotorSpeed;
        } else if (this.engineState === 'SHUTTING_DOWN') {
            const duration = this.audioManager.getShutdownDuration();
            const currentTime = this.audioManager.getShutdownCurrentTime();
            if (duration && !isNaN(duration)) {
                const progress = Math.max(0.0, 1.0 - (currentTime / duration));
                this.rotorSpeed = progress * this.maxRotorSpeed;
            } else {
                this.rotorSpeed = Math.max(0.0, this.rotorSpeed - delta * 30);
            }
        } else {
            this.rotorSpeed = 0;
        }

        for (const blade of this.rotorBlades) {
            blade.rotation.z += this.rotorSpeed * delta;
        }

        const currentPos = this.group.position;
        currentPos.y = THREE.MathUtils.lerp(currentPos.y, this.targetAltitude, delta * 3.0);

        if (this.shadowMesh && this.dropLine) {
            this.shadowMesh.position.set(currentPos.x, 0.2, currentPos.z);
            
            const heightFactor = Math.max(0.4, 1.0 - (currentPos.y / 60.0));
            this.shadowMesh.scale.set(heightFactor, heightFactor, heightFactor);

            const positions = this.dropLine.geometry.attributes.position.array;
            positions[0] = currentPos.x;
            positions[1] = 0.2;
            positions[2] = currentPos.z;

            positions[3] = currentPos.x;
            positions[4] = currentPos.y;
            positions[5] = currentPos.z;

            this.dropLine.geometry.attributes.position.needsUpdate = true;
            this.dropLine.computeLineDistances();
        }

        if (this.engineState !== 'RUNNING' || Math.abs(currentPos.y - 32) > 1.0) {
            this.targetPosition.x = currentPos.x;
            this.targetPosition.z = currentPos.z;
            this.isMoving = false;
        }

        if (this.isMoving && this.engineState === 'RUNNING') {
            const targetXZ = new THREE.Vector2(this.targetPosition.x, this.targetPosition.z);
            const currentXZ = new THREE.Vector2(currentPos.x, currentPos.z);
            const distance = currentXZ.distanceTo(targetXZ);

            if (distance > 0.3) {
                // Lower lerp value (0.8) makes acceleration much slower, heavier, and gradual
                currentXZ.lerp(targetXZ, delta * 0.8);
                
                const moveDir = new THREE.Vector2().subVectors(targetXZ, currentXZ).normalize();
                
                currentPos.x = currentXZ.x;
                currentPos.z = currentXZ.y;

                const angle = Math.atan2(moveDir.y, moveDir.x) + Math.PI;
                let normalized = (angle + Math.PI * 2) % (Math.PI * 2);
                
                const frameIndex = Math.floor((normalized / (Math.PI * 2)) * this.totalFrames + 0.5) % this.totalFrames;
                this.currentFrame = frameIndex;

                const frameOffset = this.currentFrame * (1 / this.totalFrames);
                if (this.material.map) {
                    this.material.map.offset.x = frameOffset;
                }
                if (this.material.emissiveMap) {
                    this.material.emissiveMap.offset.x = frameOffset;
                }
            } else {
                this.isMoving = false;
            }
        }
    }

    getPosition() {
        return this.group.position;
    }
}

export function setupScene() {
    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0x020d1a);
    scene.fog = new THREE.FogExp2(0x020d1a, 0.001);

    const aspect = window.innerWidth / window.innerHeight;
    const frustumSize = 80;
    
    const camera = new THREE.OrthographicCamera(
        (frustumSize * aspect) / -2, 
        (frustumSize * aspect) / 2, 
        frustumSize / 2, 
        frustumSize / -2, 
        0.1, 
        2000
    );

    camera.position.set(50, 60, 50);
    camera.lookAt(0, 0, 0);

    const renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance' });
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.0;
    document.body.appendChild(renderer.domElement);

    const ambientLight = new THREE.AmbientLight(0x6080a0, 0.8);
    scene.add(ambientLight);

    const sunLight = new THREE.DirectionalLight(0xffffff, 1.5);
    sunLight.position.set(100, 200, 100);
    scene.add(sunLight);

    window.addEventListener('resize', () => {
        const newAspect = window.innerWidth / window.innerHeight;
        camera.left = (frustumSize * newAspect) / -2;
        camera.right = (frustumSize * newAspect) / 2;
        camera.top = frustumSize / 2;
        camera.bottom = frustumSize / -2;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
    });

    return { scene, camera, renderer, sunLight, ambientLight };
}

    constructor(scene, parentGroup, playerRef) {

        this.parentGroup = parentGroup;
        this.playerRef = playerRef;

        this.particlesCount = 4000;
        this.splashCount = 1500;

            particleSize: 9.0,

            fallSpeed: 4.0,
            spread: 8.0,
            spawnRate: 10,

        this.positions = new Float32Array(this.particlesCount * 3);
        this.velocities = new Float32Array(this.particlesCount * 3);
        this.lifetimes = new Float32Array(this.particlesCount);
        this.sizes = new Float32Array(this.particlesCount);

        for (let i = 0; i < this.particlesCount; i++) {
            this.positions[i * 3] = 0;

        this.particleGeometry.setAttribute('size', new THREE.BufferAttribute(this.sizes, 1));

        // Soft feathered radial texture
        const canvas = document.createElement('canvas');
        canvas.width = 64;
        canvas.height = 64;
        const ctx = canvas.getContext('2d');
        const gradient = ctx.createRadialGradient(32, 32, 0, 32, 32, 32);
        gradient.addColorStop(0, 'rgba(235, 245, 255, 0.85)');
        gradient.addColorStop(0.3, 'rgba(200, 225, 245, 0.45)');
        gradient.addColorStop(0.7, 'rgba(175, 210, 240, 0.15)');
        gradient.addColorStop(1, 'rgba(255, 255, 255, 0)');
        ctx.fillStyle = gradient;
        ctx.fillRect(0, 0, 64, 64);

        const texture = new THREE.CanvasTexture(canvas);

        this.particleMaterial = new THREE.PointsMaterial({
            color: 0xffffff,
            size: this.params.particleSize,
            map: texture,
            transparent: true,
            blending: THREE.NormalBlending,
            depthWrite: false,
            opacity: this.params.particleOpacity
        });

        this.particleSystem = new THREE.Points(this.particleGeometry, this.particleMaterial);
        this.particleSystem.frustumCulled = false;
        this.scene.add(this.particleSystem);

        // --- Surface Impact Splash Setup ---
        this.splashGeometry = new THREE.BufferGeometry();
        this.splashPositions = new Float32Array(this.splashCount * 3);
        this.splashVelocities = new Float32Array(this.splashCount * 3);
        this.splashLifetimes = new Float32Array(this.splashCount);

        for (let i = 0; i < this.splashCount; i++) {
            this.splashPositions[i * 3] = 0;
            this.splashPositions[i * 3 + 1] = -5000;
            this.splashPositions[i * 3 + 2] = 0;
            this.splashLifetimes[i] = 0;
        }

        this.splashGeometry.setAttribute('position', new THREE.BufferAttribute(this.splashPositions, 3));

        this.splashMaterial = new THREE.PointsMaterial({
            color: 0xffffff,
            size: this.params.splashSize,
            map: texture,
            transparent: true,
            blending: THREE.NormalBlending,
            depthWrite: false,
            opacity: this.params.splashOpacity
        });

        this.splashSystem = new THREE.Points(this.splashGeometry, this.splashMaterial);
        this.splashSystem.frustumCulled = false;
        this.scene.add(this.splashSystem);
        
        this.spawnIndex = 0;
        this.splashSpawnIndex = 0;
    }

    triggerSplash(x, z, count = 1) {
        const waterLevel = 0.0;
        const splashPos = this.splashGeometry.attributes.position.array;

        for (let s = 0; s < count; s++) {
            const i = this.splashSpawnIndex;
            const idx = i * 3;

            const angle = Math.random() * Math.PI * 2;
            const radius = Math.random() * 4.5;

            splashPos[idx] = x + Math.cos(angle) * radius;
            splashPos[idx + 1] = waterLevel + 0.1;
            splashPos[idx + 2] = z + Math.sin(angle) * radius;

            const outwardSpeed = 4.0 + Math.random() * 7.0;
            this.splashVelocities[idx] = Math.cos(angle) * outwardSpeed;
            this.splashVelocities[idx + 1] = 1.5 + Math.random() * 3.0; 
            this.splashVelocities[idx + 2] = Math.sin(angle) * outwardSpeed;

            this.splashLifetimes[i] = 0.9 + Math.random() * 0.7;
            this.splashSpawnIndex = (this.splashSpawnIndex + 1) % this.splashCount;
        }
    }

    start() {
        if (this.playerRef && this.playerRef.gearDown) {
            console.log("Cannot spray water while landing gear is down! 🛞");
            this.isActive = false;
            return;
        }
        this.isActive = true;
        console.log("Water Spray: ACTIVE 💦");
    }

    stop() {
        this.isActive = false;
        console.log("Water Spray: OFF");
    }

    toggle() {
        if (this.isActive) {
            this.stop();
        } else {
            this.start();
        }
    }

    update(delta) {
        if (this.isActive && this.playerRef && this.playerRef.gearDown) {
            this.isActive = false;
            console.log("Water spray cut off: Landing gear deployed!");
        }

        const currentHeliWorldPos = new THREE.Vector3();
        this.parentGroup.getWorldPosition(currentHeliWorldPos);
        if (delta > 0) {
            this.heliVelocity.copy(currentHeliWorldPos).sub(this.lastHeliWorldPos).divideScalar(delta);
        }
        this.lastHeliWorldPos.copy(currentHeliWorldPos);

        const positions = this.particleGeometry.attributes.position.array;
        const splashPos = this.splashGeometry.attributes.position.array;
        const waterLevel = 0.0;

        // --- Update Falling Shower Particles ---
        for (let i = 0; i < this.particlesCount; i++) {
            const idx = i * 3;

            if (this.lifetimes[i] > 0) {
                positions[idx] += this.velocities[idx] * delta;
                positions[idx + 1] += this.velocities[idx + 1] * delta;
                positions[idx + 2] += this.velocities[idx + 2] * delta;

                this.velocities[idx] *= 0.93;
                this.velocities[idx + 2] *= 0.93;
                this.velocities[idx + 1] -= 9.8 * delta; 
                this.lifetimes[i] -= delta;

                if (positions[idx + 1] <= waterLevel || this.lifetimes[i] <= 0) {
                    if (positions[idx + 1] <= waterLevel) {
                        this.triggerSplash(positions[idx], positions[idx + 2], 2);
                    }
                    positions[idx + 1] = -5000;
                    this.lifetimes[i] = 0;
                }
            }
        }

        // --- Update Surface Splash Particles ---
        for (let i = 0; i < this.splashCount; i++) {
            const idx = i * 3;
            if (this.splashLifetimes[i] > 0) {
                splashPos[idx] += this.splashVelocities[idx] * delta;
                splashPos[idx + 1] += this.splashVelocities[idx + 1] * delta;
                splashPos[idx + 2] += this.splashVelocities[idx + 2] * delta;

                this.splashVelocities[idx] *= 0.88;
                this.splashVelocities[idx + 2] *= 0.88;
                this.splashVelocities[idx + 1] -= 4.5 * delta; 

                this.splashLifetimes[i] -= delta;
                if (splashPos[idx + 1] <= waterLevel || this.splashLifetimes[i] <= 0) {
                    splashPos[idx + 1] = -5000;
                    this.splashLifetimes[i] = 0;
                }
            }
        }

        // --- Handle Water Discharge Spawning ---
        if (this.isActive) {
            const spawnBatch = Math.floor(this.params.spawnRate);
            const spawnCenter = currentHeliWorldPos.clone().add(new THREE.Vector3(0, -1.8, 0));

            for (let b = 0; b < spawnBatch; b++) {
                const i = this.spawnIndex;
                const idx = i * 3;

                const angle = Math.random() * Math.PI * 2;
                const radius = 0.2 + Math.random() * 2.5;

                positions[idx] = spawnCenter.x + Math.cos(angle) * radius;
                positions[idx + 1] = spawnCenter.y;
                positions[idx + 2] = spawnCenter.z + Math.sin(angle) * radius;

                const scatter = this.params.spread;
                const trailX = -this.heliVelocity.x * 0.3;
                const trailZ = -this.heliVelocity.z * 0.3;

                this.velocities[idx] = (Math.random() - 0.5) * scatter + trailX;
                this.velocities[idx + 1] = -this.params.fallSpeed - Math.random() * 4.0; 
                this.velocities[idx + 2] = (Math.random() - 0.5) * scatter + trailZ;

                this.lifetimes[i] = 4.0; 

                this.spawnIndex = (this.spawnIndex + 1) % this.particlesCount;
            }
        }

        this.particleGeometry.attributes.position.needsUpdate = true;
        this.splashGeometry.attributes.position.needsUpdate = true;
    }
}

export class WinchSystem {
    constructor(scene, parentGroup) {
        this.scene = scene;
        this.parentGroup = parentGroup;
        this.isActive = false;

        // Winch deployment state (cable length)
        this.cableLength = 1.0;
        this.minCableLength = 1.0;
        this.maxCableLength = 12.0;

        // Create Winch Cable Line Geometry
        const lineGeo = new THREE.BufferGeometry().setFromPoints([
            new THREE.Vector3(0, -1, 0),
            new THREE.Vector3(0, -1, 0)
        ]);
        const lineMat = new THREE.LineBasicMaterial({ color: 0x222222, linewidth: 3 });
        this.cableLine = new THREE.Line(lineGeo, lineMat);
        this.cableLine.frustumCulled = false;
        this.parentGroup.add(this.cableLine);

        // Create Winch Hook / Magnet Mesh at the end of the cable
        const hookGeo = new THREE.ConeGeometry(0.4, 1.0, 6);
        const hookMat = new THREE.MeshStandardMaterial({ color: 0xffaa00, roughness: 0.3 });
        this.hookMesh = new THREE.Mesh(hookGeo, hookMat);
        this.hookMesh.rotation.x = Math.PI; // Point downwards
        this.hookMesh.position.set(0, -1, 0);
        this.parentGroup.add(this.hookMesh);

        // Initially hidden until toggled on
        this.setVisibility(false);
    }

    setVisibility(visible) {
        this.cableLine.visible = visible;
        this.hookMesh.visible = visible;
    }

    toggle() {
        this.isActive = !this.isActive;
        this.setVisibility(this.isActive);
        console.log(`Winch System: ${this.isActive ? "DEPLOYED 🪝" : "RETRACTED"}`);
    }

    stop() {
        this.isActive = false;
        this.setVisibility(false);
    }

    update(delta) {
        if (!this.isActive) return;

        // Lower or raise cable dynamically when active (or keep it stable at max deployment)
        // For now, let's make it smoothly lower down to max length when deployed
        if (this.cableLength < this.maxCableLength) {
            this.cableLength = Math.min(this.maxCableLength, this.cableLength + delta * 8.0);
        }

        // Update Hook position
        this.hookMesh.position.y = -this.cableLength;

        // Update Cable line points from helicopter bottom (-1) to hook position (-cableLength)
        const positions = this.cableLine.geometry.attributes.position.array;
        positions[1] = -1.0;                  // Start point Y (under heli)
        positions[4] = -this.cableLength;     // End point Y (at hook)
        this.cableLine.geometry.attributes.position.needsUpdate = true;
    }
}

    loadTransparentTexture(url, callback) {
        const img = new Image();
        img.src = url;
        img.onload = () => {
            const canvas = document.createElement('canvas');
            canvas.width = img.width;
            canvas.height = img.height;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0);

            const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
            const data = imgData.data;

            const bgR = data[0];
            const bgG = data[1];
            const bgB = data[2];

            for (let i = 0; i < data.length; i += 4) {
                const r = data[i];
                const g = data[i + 1];
                const b = data[i + 2];

                if (Math.abs(r - bgR) < 20 && Math.abs(g - bgG) < 20 && Math.abs(b - bgB) < 20) {
                    data[i + 3] = 0;
                }
            }

            ctx.putImageData(imgData, 0, 0);

            const texture = new THREE.CanvasTexture(canvas);
            texture.colorSpace = THREE.SRGBColorSpace;
            texture.minFilter = THREE.LinearFilter;
            texture.magFilter = THREE.LinearFilter;

            if (callback) callback(texture);
        };
    }

    createTurbine(x, z) {
        const turbineGroup = new THREE.Group();
        turbineGroup.position.set(x, 0, z);

        // 1. Turbine Static Body Mesh
        const bodyGeo = new THREE.PlaneGeometry(36, 36);
        const bodyMesh = new THREE.Mesh(bodyGeo, this.bodyMaterial);
        bodyMesh.position.set(0, 18, 0);
        turbineGroup.add(bodyMesh);

        // 2. Rotor Hub Group
        const rotorGroup = new THREE.Group();
        rotorGroup.position.set(this.config.hubX, this.config.hubY, this.config.hubZ);

            color: 0xdddddd,
            roughness: 0.3,
            side: THREE.DoubleSide
        });

        for (let i = 0; i < 3; i++) {
            const bladeGeo = new THREE.PlaneGeometry(this.config.length, this.config.thickness);
            bladeGeo.translate(this.config.length / 2, 0, 0);
            const blade = new THREE.Mesh(bladeGeo, bladeMat);
            blade.rotation.z = i * ((Math.PI * 2) / 3);
            rotorGroup.add(blade);
        }

        turbineGroup.add(rotorGroup);

        // 3. Water Foam Ring Particle System
        const count = this.foamConfig.count;
        const foamGeometry = new THREE.BufferGeometry();
        const foamPositions = new Float32Array(count * 3);
        const foamAngles = new Float32Array(count);
        const foamOffsets = new Float32Array(count);

        for (let i = 0; i < count; i++) {
            foamPositions[i * 3] = 0;
            foamPositions[i * 3 + 1] = 0;
            foamPositions[i * 3 + 2] = 0;
            foamAngles[i] = (i / count) * Math.PI * 2;
            foamOffsets[i] = Math.random() * Math.PI * 2;
        }

        foamGeometry.setAttribute('position', new THREE.BufferAttribute(foamPositions, 3));

        const canvas = document.createElement('canvas');
        canvas.width = 64;
        canvas.height = 64;
        const ctx = canvas.getContext('2d');
        const gradient = ctx.createRadialGradient(32, 32, 0, 32, 32, 32);
        gradient.addColorStop(0, 'rgba(255, 255, 255, 0.9)');
        gradient.addColorStop(0.4, 'rgba(240, 248, 255, 0.4)');
        gradient.addColorStop(1, 'rgba(255, 255, 255, 0)');
        ctx.fillStyle = gradient;
        ctx.fillRect(0, 0, 64, 64);

        const foamTexture = new THREE.CanvasTexture(canvas);
        const foamMaterial = new THREE.PointsMaterial({
            color: 0xffffff,
            size: this.foamConfig.thickness,
            map: foamTexture,
            transparent: true,
            blending: THREE.NormalBlending,
            depthWrite: false,
            opacity: 0.7

        foamParticleSystem.rotation.y = THREE.MathUtils.degToRad(this.foamConfig.rotY);
        foamParticleSystem.rotation.z = THREE.MathUtils.degToRad(this.foamConfig.rotZ);
        turbineGroup.add(foamParticleSystem);

        // 4. Red Light Mesh
        const markerGeo = new THREE.SphereGeometry(1, 16, 16);
        const markerMat = new THREE.MeshBasicMaterial({ color: 0xff0033 });
        const lightMarker = new THREE.Mesh(markerGeo, markerMat);
        lightMarker.position.set(this.lightConfig.x, this.lightConfig.y, this.lightConfig.z);
        lightMarker.scale.set(this.lightConfig.scale, this.lightConfig.scale, this.lightConfig.scale);
        turbineGroup.add(lightMarker);

        pointLight.position.set(this.lightConfig.x, this.lightConfig.y, this.lightConfig.z);

    update(delta, cameraQuaternion) {
        for (const turbine of this.turbines) {
            const elapsedTime = Date.now() * 0.001;

            turbine.rotor.rotation.z += this.config.speed * delta;

            // Water foam ring animation
            for (let i = 0; i < turbine.foamAngles.length; i++) {
                const idx = i * 3;
                const angle = turbine.foamAngles[i];
                const phase = turbine.foamOffsets[i];

                const surge = Math.sin(elapsedTime * this.foamConfig.speed + phase) * 0.5;
                const currentRadius = this.foamConfig.radius + surge;

                const x = Math.cos(angle) * currentRadius;
                const y = Math.sin(angle) * currentRadius;
                const z = Math.sin(elapsedTime * 3.0 + i) * 0.05;

                turbine.foamPositions[idx] = x;
                turbine.foamPositions[idx + 1] = y;
                turbine.foamPositions[idx + 2] = z;
            }
            turbine.foamGeometry.attributes.position.needsUpdate = true;

            // Sparks & Smoke Stack animation when active
            if (turbine.isSparking) {
                turbine.sparkLight.intensity = Math.random() > 0.3 ? 6.0 : 0.5;

                // Spark burst physics
                const sparkPos = turbine.sparkSystem.geometry.attributes.position.array;
                for (let i = 0; i < turbine.sparkLifetimes.length; i++) {
                    const idx = i * 3;
                    turbine.sparkLifetimes[i] -= delta * 1.5;

                    sparkPos[idx] += turbine.sparkVelocities[idx] * delta * (this.fxParams.sparkSpeed / 10);
                    sparkPos[idx + 1] += turbine.sparkVelocities[idx + 1] * delta * (this.fxParams.sparkSpeed / 10) - (9.8 * delta * delta);
                    sparkPos[idx + 2] += turbine.sparkVelocities[idx + 2] * delta * (this.fxParams.sparkSpeed / 10);

                    if (turbine.sparkLifetimes[i] <= 0 || sparkPos[idx + 1] < this.lightConfig.y - 2.0) {
                        sparkPos[idx] = this.lightConfig.x + (Math.random() - 0.5) * 0.5;
                        sparkPos[idx + 1] = this.lightConfig.y;
                        sparkPos[idx + 2] = this.lightConfig.z + (Math.random() - 0.5) * 0.5;

                        turbine.sparkVelocities[idx] = (Math.random() - 0.5) * 8.0;
                        turbine.sparkVelocities[idx + 1] = Math.random() * 10.0 + 3.0;
                        turbine.sparkVelocities[idx + 2] = (Math.random() - 0.5) * 8.0;

                        turbine.sparkLifetimes[i] = Math.random() * 0.8 + 0.2;
                    }
                }
                turbine.sparkSystem.geometry.attributes.position.needsUpdate = true;

                // Vertical Smoke Stack rising animation
                const stackPos = turbine.stackSystem.geometry.attributes.position.array;
                for (let i = 0; i < turbine.stackOffsets.length; i++) {
                    const idx = i * 3;
                    const phase = turbine.stackOffsets[i];

                    stackPos[idx + 1] += delta * this.fxParams.stackSpeed;
                    stackPos[idx] += Math.sin(elapsedTime * 4 + phase) * 0.02;
                    stackPos[idx + 2] += Math.cos(elapsedTime * 3 + phase) * 0.02;

                    if (stackPos[idx + 1] > this.lightConfig.y + 12.0) {
                        stackPos[idx] = this.lightConfig.x + (Math.random() - 0.5) * 0.4;
                        stackPos[idx + 1] = this.lightConfig.y + 1.0;
                        stackPos[idx + 2] = this.lightConfig.z + (Math.random() - 0.5) * 0.4;
                    }
                }
                turbine.stackSystem.geometry.attributes.position.needsUpdate = true;
            }

            if (cameraQuaternion) {
                turbine.group.quaternion.copy(cameraQuaternion);
            }
        }
    }
}
